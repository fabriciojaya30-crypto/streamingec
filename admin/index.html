async function subirImagen(id) { const input = document.getElementById(`file-${id}`); if (!input) { alert("No se encontró el selector de imagen."); return; } const archivo = input.files[0]; if (!archivo) { mostrarEstado( "❌ Selecciona primero una imagen.", true ); return; } if (!archivo.type.startsWith("image/")) { mostrarEstado( "❌ El archivo seleccionado no es una imagen.", true ); return; } if (archivo.size > 5 * 1024 * 1024) { mostrarEstado( "❌ La imagen supera los 5 MB.", true ); return; } try { mostrarEstado( "⬆️ Subiendo imagen..." ); // IMPORTANTE: // El backend /api/upload.js // espera multipart/form-data const formulario = new FormData(); formulario.append( "file", archivo ); const respuesta = await fetch( "/api/upload", { method: "POST", body: formulario } ); const texto = await respuesta.text(); let resultado; try { resultado = JSON.parse(texto); } catch { throw new Error( "El servidor devolvió una respuesta inválida." ); } if (!respuesta.ok) { throw new Error( resultado.detalle || resultado.error || "No se pudo subir la imagen." ); } if (!resultado.url) { throw new Error( "El servidor no devolvió la URL de la imagen." ); } // Guardar URL en el producto const producto = productos.find( p => Number(p.id) === Number(id) ); if (producto) { producto.imagen = resultado.url; } // Guardar URL en el campo oculto const campoImagen = document.getElementById( `imagen-${id}` ); if (campoImagen) { campoImagen.value = resultado.url; } // Mostrar vista previa const preview = document.getElementById( `preview-${id}` ); if (preview) { if (preview.tagName === "IMG") { preview.src = resultado.url; } else { preview.outerHTML = ` `; } } mostrarEstado( "✅ Imagen subida correctamente." ); } catch (error) { console.error( "ERROR IMAGEN:", error ); mostrarEstado( "❌ Error al subir imagen: " + error.message, true ); } }
